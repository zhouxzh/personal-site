### 3.3 完整用户交互案例

```python bmi_calculator.py
"""
BMI计算器 v1.0
BMI = 体重kg / (身高m)^2
标准范围：
  低于18.5 → 体重过轻
  18.5-24 → 正常
  25-28 → 过重
  28-32 → 肥胖
"""
print("=" * 20)
height = float(input("输入身高(m)："))
weight = float(input("输入体重(kg)："))

bmi = weight / height**2
print(f"\nBMI计算结果: {bmi:.1f}")  # 保留1位小数

if bmi < 18.5:
    status = "过轻"
elif 18.5 <= bmi < 24:
    status = "正常"
elif 24 <= bmi < 28:
    status = "过重"
else:
    status = "肥胖"

print(f"健康状态: {status}")
